const API_URL = "https://script.google.com/macros/s/AKfycby2WqjXjLIF61w7FtuGLTmHXnGC1e9C42d8szEXrcJEn5bFUGBXWkYFfgDaZ6wzr48Iqg/exec"; let scanner = null; let scannerAktif = false; let sedangProses = false; let requestScript = null; let requestTimeout = null; // MENAMPILKAN PESAN function tampilkan(pesan, status = "") { const box = document.getElementById("hasil"); box.className = "hasil " + status; box.innerText = pesan; } // MEMULAI SCANNER function mulaiScan() { if (scannerAktif) return; if (!window.isSecureContext || !navigator.mediaDevices) { tampilkan( "Browser tidak mendukung akses kamera. Pastikan halaman dibuka melalui HTTPS.", "gagal" ); return; } if (typeof Html5Qrcode === "undefined") { tampilkan( "Library QR Scanner gagal dimuat. Periksa koneksi internet dan muat ulang halaman.", "gagal" ); return; } tampilkan("Meminta izin kamera..."); const btnMulai = document.getElementById("btnMulai"); btnMulai.disabled = true; scanner = new Html5Qrcode("reader"); scanner.start( { facingMode: "environment" }, { fps: 10, qrbox: { width: 230, height: 230 }, aspectRatio: 1.333333 }, scanBerhasil ).then(function() { scannerAktif = true; btnMulai.hidden = true; btnMulai.disabled = false; document.getElementById("btnStop").hidden = false; tampilkan( "Kamera aktif. Arahkan ke QR Code undangan wisudawan." ); }).catch(function(error) { scannerAktif = false; btnMulai.disabled = false; tampilkan( "Gagal mengaktifkan kamera.\n\n" + error + "\n\nPeriksa izin kamera browser dan coba kembali.", "gagal" ); }); } // KETIKA QR BERHASIL TERBACA function scanBerhasil(teks) { if (sedangProses) return; sedangProses = true; scanner.pause(true); tampilkan("Memeriksa identitas peserta..."); kirimKeServer(teks.trim()); } // MENGIRIM TOKEN KE APPS SCRIPT function kirimKeServer(token) { bersihkanRequest(); const callbackName = "absensiCallback"; window[callbackName] = function(data) { bersihkanRequest(); tampilkanHasil(data); }; const separator = API_URL.includes("?") ? "&" : "?"; const url = API_URL + separator + "action=scan" + "&token=" + encodeURIComponent(token) + "&callback=" + callbackName + "&t=" + Date.now(); requestScript = document.createElement("script"); requestScript.src = url; requestScript.onerror = function() { bersihkanRequest(); tampilkan( "Koneksi ke server gagal. Periksa internet dan coba kembali.", "gagal" ); document.getElementById("btnOK").style.display = "block"; }; document.body.appendChild(requestScript); requestTimeout = setTimeout(function() { bersihkanRequest(); tampilkan( "Server tidak merespons. Silakan coba kembali.", "gagal" ); document.getElementById("btnOK").style.display = "block"; }, 20000); } // MEMBERSIHKAN REQUEST function bersihkanRequest() { clearTimeout(requestTimeout); requestTimeout = null; if (requestScript) { requestScript.remove(); requestScript = null; } delete window.absensiCallback; } // MENAMPILKAN HASIL ABSENSI function tampilkanHasil(data) { const btn = document.getElementById("btnOK"); if (data.sukses) { tampilkan( "✓ BERHASIL HADIR\n\n" + "Nama: " + data.nama + "\nNIM: " + data.nim + "\nFakultas: " + data.fakultas + "\nWaktu: " + data.waktu, "sukses" ); } else if (data.duplikat) { tampilkan( "⚠ SUDAH ABSENSI\n\n" + "Nama: " + data.nama + "\nNIM: " + data.nim + "\nPeserta telah tercatat hadir sebelumnya.", "gagal" ); } else { tampilkan( "✕ ABSENSI GAGAL\n\n" + data.pesan, "gagal" ); } // Hasil tetap tampil sampai tombol OK ditekan btn.style.display = "block"; } // MELANJUTKAN SCAN SETELAH TOMBOL OK function lanjutScan() { document.getElementById("btnOK").style.display = "none"; tampilkan( "Arahkan kamera ke QR Code undangan wisudawan." ); sedangProses = false; if (scannerAktif && scanner) { scanner.resume(); } } // MENGHENTIKAN SCANNER function berhentiScan() { if (!scanner || !scannerAktif) return; scanner.stop().then(function() { scannerAktif = false; scanner.clear(); scanner = null; document.getElementById("reader").innerHTML = ""; document.getElementById("btnMulai").hidden = false; document.getElementById("btnMulai").disabled = false; document.getElementById("btnStop").hidden = true; document.getElementById("btnOK").style.display = "none"; sedangProses = false; tampilkan( "Kamera dihentikan. Tekan MULAI SCAN untuk memulai kembali." ); }).catch(function(error) { tampilkan( "Gagal menghentikan kamera: " + error, "gagal" ); }); }
